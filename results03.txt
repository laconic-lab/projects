errors = []

for root in roots:
    if not root.is_dir():
        errors.append(f"Missing scan root: {root}")
        continue

    for directory in [root, *(
        p for p in root.rglob("*")
        if p.is_dir() and not any(
            part in excluded for part in p.parts
        )
    )]:
        if any(part in excluded for part in directory.parts):
            continue
        try:
            for path in directory.iterdir():
                if path.is_file() and path.name.lower().endswith(suffixes):
                    matches.append(path)
        except OSError as exc:
            errors.append(f"Cannot inspect {directory}: {exc}")

matches = sorted(set(matches), key=lambda p: str(p).lower())

for path in matches:

    print(f"PATH: {path}")
    print(f"SIZE: {path.stat().st_size} bytes")

        content = path.read_text(encoding="utf-8")
        print(content, end="" if content.endswith("\n") else "\n")
    except (OSError, UnicodeError) as exc:
        print(f"READ ERROR: {exc}")
