## A1 — Host and user-namespace capabilities

STEAM DECK — Konsole / existing SSH session

Run this complete block:

```
echo "=== HOST IDENTITY ==="
id
cat /etc/os-release | grep -E '^(NAME|VERSION|ID)='
uname -r

echo
echo "=== SUBORDINATE UID/GID CONFIGURATION ==="
for f in /etc/subuid /etc/subgid; do
    echo "--- $f ---"
    if [ -r "$f" ]; then
        grep -E '^deck:' "$f" || echo "No deck entry"
    else
        echo "Not present or not readable"
    fi
done

echo
echo "=== USER NAMESPACE SUPPORT ==="
for key in \
    user.max_user_namespaces \
    kernel.unprivileged_userns_clone \
    kernel.apparmor_restrict_unprivileged_userns
do
    sysctl "$key" 2>&1
done

echo
echo "=== UID/GID HELPER AVAILABILITY ==="
for cmd in newuidmap newgidmap; do
    command -v "$cmd" || echo "$cmd: not found"
done
```

