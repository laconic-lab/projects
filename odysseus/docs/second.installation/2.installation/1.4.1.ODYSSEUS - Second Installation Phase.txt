```
echo "=================================================="
echo " ODYSSEUS 1.4 — STAGE 1 ENVIRONMENT AUDIT"
echo " READ-ONLY DIAGNOSTICS"
echo "=================================================="

echo
echo "=== 1. STEAMOS HOST ==="
cat /etc/os-release | grep -E '^(NAME|VERSION|ID)='
uname -r
id
hostname

echo
echo "=== 2. HARDWARE AND STORAGE ==="
lscpu | grep -E '^(Architecture|CPU\(s\)|Model name):'
free -h
df -h /home

echo
echo "=== 3. RPM-OSTREE STATUS ==="
rpm-ostree status

echo
echo "=== 4. DISTROBOX ==="
command -v distrobox
distrobox --version
distrobox list

echo
echo "=== 5. PODMAN ==="
command -v podman
podman --version
podman info --format 'Rootless={{.Host.Security.Rootless}} GraphRoot={{.Store.GraphRoot}}'
podman ps -a --format "table {{.ID}}\t{{.Names}}\t{{.Image}}\t{{.Status}}\t{{.Ports}}"

echo
echo "=== 6. ODYSSEUS REMNANTS ==="
find /home/deck -maxdepth 2 -iname '*odysseus*' \
    -printf '%y %m %u:%g %p\n' 2>/dev/null

echo
echo "=== 7. EXPECTED PROJECT DIRECTORIES ==="
for p in \
    /home/deck/OdysseusProject \
    /home/deck/OdysseusRecovery \
    /home/deck/.config/odysseus \
    /home/deck/.config/odysseus/providers
do
    if [ -e "$p" ]; then
        stat -c '%A %a %U:%G %n' "$p"
    else
        echo "ABSENT: $p"
    fi
done

echo
echo "=== 8. PROVIDER CREDENTIAL METADATA ==="
if [ -d /home/deck/.config/odysseus/providers ]; then
    find /home/deck/.config/odysseus/providers \
        -maxdepth 3 -type f \
        -printf '%M %u:%g %s bytes %p\n'
else
    echo "Provider directory absent."
fi

echo
echo "=== 9. EXISTING ODYSSEUS USER SERVICES ==="
systemctl --user list-unit-files --no-pager 2>/dev/null \
    | grep -i odysseus || echo "No matching user service units."

echo
echo "=================================================="
echo " END OF READ-ONLY AUDIT"
echo "=================================================="
```
